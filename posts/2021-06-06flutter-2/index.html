    void main(List<String> arguments) {
  //(x) {return x + 5;} is an anonymous function
  //twicePlusFive is also a function
  final twicePlusFive = twice((x) {return x + 5;});
  print(twicePlusFive(3)); //output is 13.
}
//(return type) Function(param type)
int Function(int) twice(int Function(int) f) {
  return (int x) {return f(f(x));};
}
